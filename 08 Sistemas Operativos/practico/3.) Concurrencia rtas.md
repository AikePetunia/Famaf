### 4. Dekker (para 2 procesos: $P_0$ y $P_1$)
```C
int wants_to_enter[2] = {0, 0};
int turn = 0;

// Proceso Pi (donde self = i, other = 1 - i)
void lock(int self) {
    int other = 1 - self;
    wants_to_enter[self] = 1;
    while (wants_to_enter[other] == 1) {
        if (turn != self) {
            wants_to_enter[self] = 0;
            while (turn != self)
                ; // spin-wait
            wants_to_enter[self] = 1;
        }
    }
}

void unlock(int self) {
    int other = 1 - self;
    turn = other;
    wants_to_enter[self] = 0;
}
```

### 5. Peterson (para 2 procesos: $P_0$ y $P_1$)
```c
int flag[2] = {0, 0};
int turn = 0;

// Proceso Pi (donde self = i, other = 1 - i)
void lock(int self) {
    int other = 1 - self;
    flag[self] = 1;
    turn = other;
    while (flag[other] == 1 && turn == other)
        ; // spin-wait
}

void unlock(int self) {
    flag[self] = 0;
}
```

### 7. LL-SC (Load-Linked / Store-Conditional)
```c
typedef struct __lock_t {
    int flag;

    lock->flag = 0;
}

void lock(lock_t *lock) {
    while (1) {
        while (LoadLinked(&lock->flag) == 1)
            ; // spin-wait
```

### 8. Fetch-And-Add (Ticket Lock)
```C
typedef struct __lock_t {
    int ticket;
    int turn;
} lock_t;

void init(lock_t *lock) {
    lock->ticket = 0;
    lock->turn = 0;
}

void lock(lock_t *lock) {
    int myturn = FetchAndAdd(&lock->ticket);
    while (lock->turn != myturn)
        ; // spin-wait
}

void unlock(lock_t *lock) {
    lock->turn = lock->turn + 1;
}
```

### 10. Test-and-Set with Park (TS-With-Park / Solaris style con cola)

```C
typedef struct __lock_t {
    int flag;
    int guard;
    queue_t *q;
} lock_t;

void init(lock_t *m) {
    m->flag = 0;
    m->guard = 0;
    queue_init(m->q);
}

void lock(lock_t *m) {
    while (TestAndSet(&m->guard, 1) == 1)
        ; // adquiere el spinlock interno de guardia
    
    if (m->flag == 0) {
        m->flag = 1; // adquiere el lock
        m->guard = 0;
    } else {
        queue_add(m->q, gettid());
        m->guard = 0;
        park(); // pone el hilo a dormir
    }
}

void unlock(lock_t *m) {
    while (TestAndSet(&m->guard, 1) == 1)
        ; // adquiere el spinlock interno de guardia
    
    if (queue_empty(m->q)) {
        m->flag = 0; // libera el lock directamente
    } else {
        unpark(queue_remove(m->q)); // despierta al siguiente proceso
    }
    m->guard = 0;
}
```
